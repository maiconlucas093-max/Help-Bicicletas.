Seu carrinho está vazio.

ESCOLHER PRODUTOS `; totalElemento.textContent = formatarPreco(0); return; } lista.innerHTML = itens.map(item => `
${escaparTextoCheckout(item.nome)}

Quantidade: ${item.quantidade}

${formatarPreco(item.preco * item.quantidade)}
`).join(""); totalElemento.textContent = formatarPreco(calcularTotal()); } // Protege os nomes dos produtos ao exibi-los function escaparTextoCheckout(valor) { const elemento = document.createElement("span"); elemento.textContent = valor; return elemento.innerHTML; } // Controla os campos conforme a entrega escolhida function prepararTipoEntrega() { const tipoEntrega = document.getElementById("tipo-entrega"); const camposEndereco = document.getElementById("campos-endereco"); if (!tipoEntrega || !camposEndereco) return; const camposObrigatorios = [ "cep-cliente", "endereco-cliente", "numero-cliente", "cidade-cliente" ]; function atualizarEndereco() { const entregaSelecionada = tipoEntrega.value === "entrega"; camposEndereco.hidden = !entregaSelecionada; camposObrigatorios.forEach(id => { const campo = document.getElementById(id); if (campo) { campo.required = entregaSelecionada; } }); } tipoEntrega.addEventListener("change", atualizarEndereco); atualizarEndereco(); } // Validação inicial do formulário function prepararFormularioCheckout() { const formulario = document.getElementById("form-checkout"); if (!formulario) return; formulario.addEventListener("submit", evento => { evento.preventDefault(); if (obterCarrinho().length === 0) { alert("Seu carrinho está vazio.
